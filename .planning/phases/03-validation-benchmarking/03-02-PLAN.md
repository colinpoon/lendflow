1. Define pricing constants (as of 2026-02):

```typescript
// Claude Sonnet 4 (claude-sonnet-4-20250514)
export const VISION_COST_PER_M_INPUT = 3.00;   // USD per million input tokens
export const VISION_COST_PER_M_OUTPUT = 15.00; // USD per million output tokens

// GPT-4 Turbo (gpt-4-turbo-2024-04-09)
export const TEXT_COST_PER_M_INPUT = 10.00;
export const TEXT_COST_PER_M_OUTPUT = 30.00;
```

2. Define interfaces:
```typescript
export interface TokenUsage {
  input_tokens: number;
  output_tokens: number;
  model: string;
}

export interface CostBreakdown {
  input_cost: number;
  output_cost: number;
  total_cost: number;
  model: string;
}
```

3. Implement cost calculation functions:
```typescript
export function calculateVisionCost(usage: TokenUsage): CostBreakdown {
  const inputCost = (usage.input_tokens / 1_000_000) * VISION_COST_PER_M_INPUT;
  const outputCost = (usage.output_tokens / 1_000_000) * VISION_COST_PER_M_OUTPUT;
  return {
    input_cost: inputCost,
    output_cost: outputCost,
    total_cost: inputCost + outputCost,
    model: usage.model,
  };
}

export function calculateTextCost(usage: TokenUsage): CostBreakdown {
  const inputCost = (usage.input_tokens / 1_000_000) * TEXT_COST_PER_M_INPUT;
  const outputCost = (usage.output_tokens / 1_000_000) * TEXT_COST_PER_M_OUTPUT;
  return {
    input_cost: inputCost,
    output_cost: outputCost,
    total_cost: inputCost + outputCost,
    model: usage.model,
  };
}
```

4. Export a unified cost calculator:
```typescript
export function calculateCost(usage: TokenUsage, pipeline: 'vision' | 'text'): CostBreakdown {
  return pipeline === 'vision'
    ? calculateVisionCost(usage)
    : calculateTextCost(usage);
}
```
