/**
 * Get ground truth by filename (partial match) and fiscal year.
 * Matching strategy: Case-insensitive partial match on filename.
 * Example: "zedcor" matches "2024-12-31-Q4-Zedcor-Inc.-Financial-Stmts-4.9.2025v1.pdf"
 */
export function getGroundTruth(
  filename: string,
  fiscalYear: string
): GroundTruthEntry | undefined {
  const normalizedInput = filename.toLowerCase();
  return GROUND_TRUTH.find(
    (entry) =>
      entry.fiscal_year === fiscalYear &&
      (entry.filename.toLowerCase().includes(normalizedInput) ||
       normalizedInput.includes(entry.filename.toLowerCase()) ||
       // Also match by document identifier (e.g., "Zedcor-FY2024")
       normalizedInput.includes(entry.document.toLowerCase().replace('-', '')))
  );
}
```

  metric: string;

  vision_accurate: boolean;
}
```

3. Define `AccuracySummary` interface:
```typescript
export interface AccuracySummary {
  document: string;
  fiscal_year: string;
  results: AccuracyResult[];
  text_accuracy_pct: number;   // % of metrics that are accurate
  vision_accuracy_pct: number;
  winner: 'text' | 'vision' | 'tie';
}
```

4. Implement `calculateAccuracy` function:
```typescript
export function calculateAccuracy(
  truth: GroundTruthEntry,
  textMetrics: ComputedMetrics | null,
  visionMetrics: ComputedMetrics | null
): AccuracySummary
```

Logic:
- For each metric in truth.values that is not null/undefined:
  - Get corresponding value from textMetrics and visionMetrics
  - Calculate variance: abs((extracted - truth) / truth) * 100
  - Determine accuracy: variance < ACCURACY_THRESHOLD_PCT
- Calculate overall accuracy percentage for each pipeline
- Determine winner based on which has higher accuracy_pct

Handle edge cases:
- If truth value is 0, use absolute difference instead of percentage
- If extracted value is null, that metric is NOT accurate
- If both pipelines have same accuracy_pct, winner is 'tie'
